   389→
   390→@login_required
   391→def reports_view(request):
   392→    report_type = request.GET.get('type', 'overview')
   393→    period = request.GET.get('period', 'month')
   394→
   395→    today = timezone.now().date()
   396→    if period == 'week':
   397→        start_date = today - timedelta(days=7)
   398→    elif period == 'month':
   399→        start_date = today - timedelta(days=30)
   400→    elif period == 'quarter':
   401→        start_date = today - timedelta(days=90)
   402→    elif period == 'year':
   403→        start_date = today - timedelta(days=365)
   404→    else:
   405→        start_date = today - timedelta(days=30)
   406→
   407→    firs_in_period = FIR.objects.filter(created_at__date__gte=start_date)
   408→    total_firs_period = firs_in_period.count()
   409→    resolved = firs_in_period.filter(status='closed').count()
   410→    resolution_rate = round((resolved / total_firs_period * 100), 1) if total_firs_period > 0 else 0
   411→
   412→    category_stats = firs_in_period.values('crime_category').annotate(
   413→        count=Count('id'),
   414→        closed_count=Count('id', filter=Q(status='closed'))
   415→    ).order_by('-count')
   416→
   417→    monthly_trend = FIR.objects.filter(
   418→        created_at__date__gte=start_date
   419→    ).annotate(
   420→        month=TruncMonth('created_at')
   421→    ).values('month').annotate(
   422→        total=Count('id'),
   423→        closed=Count('id', filter=Q(status='closed'))
   424→    ).order_by('month')
   425→
   426→    priority_stats = firs_in_period.values('priority').annotate(count=Count('id')).order_by('-count')
   427→
   428→    avg_attendance = Attendance.objects.filter(date__gte=start_date, status='present').count()
   429→    total_leave_days = LeaveRequest.objects.filter(
   430→        status='approved', start_date__gte=start_date
   431→    ).annotate(days=F('end_date') - F('start_date')).aggregate(total=Sum('days'))
   432→
   433→    total_fuel_cost = FuelLog.objects.filter(date__gte=start_date).aggregate(
   434→        total=Sum(F('liters') * F('cost_per_liter'))
   435→    )['total'] or 0
   436→
   437→    vehicles_in_maintenance = Vehicle.objects.filter(status='maintenance').count()
   438→
   439→    context = {
   440→        'report_type': report_type, 'period': period,
   441→        'start_date': start_date, 'today': today,
   442→        'total_firs_period': total_firs_period, 'resolved': resolved,
   443→        'resolution_rate': resolution_rate,
   444→        'category_stats': list(category_stats),
   445→        'monthly_trend': list(monthly_trend),
   446→        'priority_stats': list(priority_stats),
   447→        'total_fuel_cost': total_fuel_cost,
   448→        'vehicles_in_maintenance': vehicles_in_maintenance,
   449→        'page_title': 'Reports & Analytics',
   450→    }
   451→    return render(request, 'reports/reports.html', context)

   497→    if type_filter:

   592→    vehicle = get_object_or_404(Vehicle, pk=vehicle_id)

   599→        messages.warning(request, f'{vehicle.vehicle_number} is already assigned to {active_assignment.officer.full_name}. Return it first before reassigning.')
   600→        return redirect('vehicle_detail', vehicle_id=vehicle_id)
   601→
   602→    form = VehicleAssignmentForm(request.POST)
   603→    if form.is_valid():
   604→        assignment = form.save(commit=False)
   605→        assignment.vehicle = vehicle
   606→        assignment.save()
   607→        vehicle.status = 'in_use'
   608→        vehicle.mileage = assignment.odometer_start or vehicle.mileage
   609→        vehicle.save()
   610→        messages.success(request, f'{vehicle.vehicle_number} assigned to {assignment.officer.full_name} successfully.')
   611→    else:
   612→        for field, errors in form.errors.items():
   613→            for error in errors:
   614→                messages.error(request, f'{field}: {error}')
   615→    return redirect('vehicle_detail', vehicle_id=vehicle_id)
