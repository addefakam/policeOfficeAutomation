    27→    if request.user.is_authenticated:
    28→        return redirect('dashboard')
    29→    if request.method == 'POST':
    30→        form = CustomLoginForm(request, data=request.POST)
    31→        if form.is_valid():
    32→            user = form.get_user()
    33→            login(request, user)
    34→            next_url = request.GET.get('next', '/dashboard/')
    35→            return redirect(next_url)

    37→        form = CustomLoginForm()
    38→    return render(request, 'base/login.html', {'form': form})

   382→
   383→@login_required
   384→def reports_view(request):
   385→    report_type = request.GET.get('type', 'overview')
   386→    period = request.GET.get('period', 'month')
   387→
   388→    today = timezone.now().date()
   389→    if period == 'week':
   390→        start_date = today - timedelta(days=7)
   391→    elif period == 'month':
   392→        start_date = today - timedelta(days=30)
   393→    elif period == 'quarter':
   394→        start_date = today - timedelta(days=90)
   395→    elif period == 'year':
   396→        start_date = today - timedelta(days=365)
   397→    else:
   398→        start_date = today - timedelta(days=30)
   399→
   400→    firs_in_period = FIR.objects.filter(created_at__date__gte=start_date)
   401→    total_firs_period = firs_in_period.count()
   402→    resolved = firs_in_period.filter(status='closed').count()
   403→    resolution_rate = round((resolved / total_firs_period * 100), 1) if total_firs_period > 0 else 0
   404→
   405→    category_stats = firs_in_period.values('crime_category').annotate(
   406→        count=Count('id'),
   407→        closed_count=Count('id', filter=Q(status='closed'))
   408→    ).order_by('-count')
   409→
   410→    monthly_trend = FIR.objects.filter(
   411→        created_at__date__gte=start_date
   412→    ).annotate(
   413→        month=TruncMonth('created_at')
   414→    ).values('month').annotate(
   415→        total=Count('id'),
   416→        closed=Count('id', filter=Q(status='closed'))
   417→    ).order_by('month')
   418→
   419→    priority_stats = firs_in_period.values('priority').annotate(count=Count('id')).order_by('-count')
   420→
   421→    avg_attendance = Attendance.objects.filter(date__gte=start_date, status='present').count()
   422→    total_leave_days = LeaveRequest.objects.filter(
   423→        status='approved', start_date__gte=start_date
   424→    ).annotate(days=F('end_date') - F('start_date')).aggregate(total=Sum('days'))
   425→
   426→    total_fuel_cost = FuelLog.objects.filter(date__gte=start_date).aggregate(
   427→        total=Sum(F('liters') * F('cost_per_liter'))
   428→    )['total'] or 0
   429→
   430→    vehicles_in_maintenance = Vehicle.objects.filter(status='maintenance').count()
   431→
   432→    context = {
   433→        'report_type': report_type, 'period': period,
   434→        'start_date': start_date, 'today': today,
   435→        'total_firs_period': total_firs_period, 'resolved': resolved,
   436→        'resolution_rate': resolution_rate,
   437→        'category_stats': list(category_stats),
   438→        'monthly_trend': list(monthly_trend),
   439→        'priority_stats': list(priority_stats),
   440→        'total_fuel_cost': total_fuel_cost,
   441→        'vehicles_in_maintenance': vehicles_in_maintenance,
   442→        'page_title': 'Reports & Analytics',
   443→    }
   444→    return render(request, 'reports/reports.html', context)

   481→@login_required
   482→def vehicle_list_view(request):
   483→    vehicles = Vehicle.objects.all()
   484→    status_filter = request.GET.get('status', '')
   485→    type_filter = request.GET.get('type', '')
   486→    search_query = request.GET.get('q', '')
   487→
   488→    if status_filter:
   489→        vehicles = vehicles.filter(status=status_filter)
   490→    if type_filter:
   491→        vehicles = vehicles.filter(vehicle_type=type_filter)
   492→    if search_query:
   493→        vehicles = vehicles.filter(
   494→            Q(vehicle_number__icontains=search_query) |
   495→            Q(make__icontains=search_query) |
   496→            Q(model__icontains=search_query)
   497→        )
   498→
   499→    status_counts = {s: Vehicle.objects.filter(status=s).count() for s, _ in Vehicle.STATUS_CHOICES}
   500→
   501→    active_assignments = VehicleAssignment.objects.filter(
   502→        vehicle=OuterRef('pk'), returned_date__isnull=True
   503→    ).select_related('officer').order_by('-assigned_date')
   504→
   505→    vehicles = vehicles.annotate(
   506→        _assignee_name=Subquery(active_assignments.values('officer__full_name')[:1]),
   507→        _assignee_id=Subquery(active_assignments.values('officer__id')[:1]),
   508→        _assigned_date=Subquery(active_assignments.values('assigned_date')[:1]),
   509→    )

   593→        return redirect('vehicle_detail', vehicle_id=vehicle_id)
   594→
   595→    form = VehicleAssignmentForm(request.POST)
   596→    if form.is_valid():
   597→        assignment = form.save(commit=False)
   598→        assignment.vehicle = vehicle
   599→        assignment.save()
   600→        vehicle.status = 'in_use'
   601→        vehicle.mileage = assignment.odometer_start or vehicle.mileage
   602→        vehicle.save()
   603→        messages.success(request, f'{vehicle.vehicle_number} assigned to {assignment.officer.full_name} successfully.')
   604→    else:
   605→        for field, errors in form.errors.items():
   606→            for error in errors:
   607→                messages.error(request, f'{field}: {error}')
   608→    return redirect('vehicle_detail', vehicle_id=vehicle_id)
