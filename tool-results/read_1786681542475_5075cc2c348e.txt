   388→   388→@login_required
   389→   389→def reports_view(request):
   390→   390→    report_type = request.GET.get('type', 'overview')
   391→   391→    period = request.GET.get('period', 'month')
   392→   392→
   393→   393→    # Base date range
   394→   394→    today = timezone.now().date()
   395→   395→    if period == 'week':
   396→   396→        start_date = today - timedelta(days=7)
   397→   397→    elif period == 'month':
   398→   398→        start_date = today - timedelta(days=30)
   399→   399→    elif period == 'quarter':
   400→   400→        start_date = today - timedelta(days=90)
   401→   401→    elif period == 'year':
   402→   402→        start_date = today - timedelta(days=365)
   403→   403→    else:
   404→   404→        start_date = today - timedelta(days=30)
   405→   405→
   406→   406→    # FIR Statistics
   407→   407→    firs_in_period = FIR.objects.filter(created_at__date__gte=start_date)
   408→   408→    total_firs_period = firs_in_period.count()
   409→   409→    resolved = firs_in_period.filter(status='closed').count()
   410→   410→    resolution_rate = round((resolved / total_firs_period * 100), 1) if total_firs_period > 0 else 0
   411→   411→
   412→   412→    # Crime category breakdown
   413→   413→    category_stats = firs_in_period.values('crime_category').annotate(
   414→   414→        count=Count('id'),
   415→   415→        closed_count=Count('id', filter=Q(status='closed'))
   416→   416→    ).order_by('-count')
   417→   417→
   418→   418→    # Monthly trend
   419→   419→    monthly_trend = FIR.objects.filter(
   420→   420→        created_at__date__gte=start_date
   421→   421→    ).annotate(
   422→   422→        month=TruncMonth('created_at')
   423→   423→    ).values('month').annotate(
   424→   424→        total=Count('id'),
   425→   425→        closed=Count('id', filter=Q(status='closed'))
   426→   426→    ).order_by('month')
   427→   427→
   428→   428→    # Priority distribution
   429→   429→    priority_stats = firs_in_period.values('priority').annotate(count=Count('id')).order_by('-count')
   430→   430→
   431→   431→    # Staff statistics
   432→   432→    avg_attendance = Attendance.objects.filter(date__gte=start_date, status='present').count()
   433→   433→    total_leave_days = LeaveRequest.objects.filter(
   434→   434→        status='approved', start_date__gte=start_date
   435→   435→    ).annotate(days=F('end_date') - F('start_date')).aggregate(total=Sum('days'))
   436→   436→
   437→   437→    # Vehicle statistics
   438→   438→    total_fuel_cost = FuelLog.objects.filter(date__gte=start_date).aggregate(
   439→   439→        total=Sum(F('liters') * F('cost_per_liter'))
