   117→// ========== HELPERS ==========
   118→const fmt = (d?: string) => {
   119→  if (!d) return '—'
   120→  try { return new Date(d).toLocaleDateString('en-US', { year: 'numeric', month: 'short', day: 'numeric' }) } catch { return d }
   121→}
   122→const fmtDT = (d?: string) => {
   123→  if (!d) return '—'
   124→  try { return new Date(d).toLocaleString('en-US', { year: 'numeric', month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' }) } catch { return d }
   125→}
   126→const fmtTime = (d?: string) => {
   127→  if (!d) return '—'
   128→  try { return new Date(d).toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit' }) } catch { return d }

   338→  const [loading, setLoading] = useState(true)

   342→  }, [])
   343→
   344→  if (loading) return <div className="space-y-4"><div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">{Array.from({length:8}).map((_,i)=><Card key={i}><CardContent className="p-4 h-24 bg-slate-100 animate-pulse rounded-lg"/></Card>)}</div></div>

   346→
   347→  const maxCrimeCount = Math.max(...(data.crimeByCategory.map(c => c.count) || [1]), 1)
   348→
   349→  return (
