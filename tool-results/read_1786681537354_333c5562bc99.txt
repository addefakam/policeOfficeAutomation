   387→
   388→@login_required
   389→def reports_view(request):
   390→    report_type = request.GET.get('type', 'overview')
   391→    period = request.GET.get('period', 'month')
   392→
   393→    # Base date range
   394→    today = timezone.now().date()
   395→    if period == 'week':
   396→        start_date = today - timedelta(days=7)
   397→    elif period == 'month':
   398→        start_date = today - timedelta(days=30)
   399→    elif period == 'quarter':
   400→        start_date = today - timedelta(days=90)
   401→    elif period == 'year':
   402→        start_date = today - timedelta(days=365)
   403→    else:
   404→        start_date = today - timedelta(days=30)
   405→
   406→    # FIR Statistics
   407→    firs_in_period = FIR.objects.filter(created_at__date__gte=start_date)
   408→    total_firs_period = firs_in_period.count()
   409→    resolved = firs_in_period.filter(status='closed').count()
   410→    resolution_rate = round((resolved / total_firs_period * 100), 1) if total_firs_period > 0 else 0
   411→
   412→    # Crime category breakdown
   413→    category_stats = firs_in_period.values('crime_category').annotate(
   414→        count=Count('id'),
   415→        closed_count=Count('id', filter=Q(status='closed'))
   416→    ).order_by('-count')
   417→
   418→    # Monthly trend
   419→    monthly_trend = FIR.objects.filter(
   420→        created_at__date__gte=start_date
   421→    ).annotate(
   422→        month=TruncMonth('created_at')
   423→    ).values('month').annotate(
   424→        total=Count('id'),
   425→        closed=Count('id', filter=Q(status='closed'))
   426→    ).order_by('month')
   427→
   428→    # Priority distribution
   429→    priority_stats = firs_in_period.values('priority').annotate(count=Count('id')).order_by('-count')
   430→
   431→    # Staff statistics
   432→    avg_attendance = Attendance.objects.filter(date__gte=start_date, status='present').count()
   433→    total_leave_days = LeaveRequest.objects.filter(
   434→        status='approved', start_date__gte=start_date
   435→    ).annotate(days=F('end_date') - F('start_date')).aggregate(total=Sum('days'))
   436→
   437→    # Vehicle statistics
   438→    total_fuel_cost = FuelLog.objects.filter(date__gte=start_date).aggregate(
   439→        total=Sum(F('liters') * F('cost_per_liter'))
   440→    )['total'] or 0
   441→
   442→    vehicles_in_maintenance = Vehicle.objects.filter(status='maintenance').count()
   443→
   444→    context = {
   445→        'report_type': report_type, 'period': period,
   446→        'start_date': start_date, 'today': today,
   447→        'total_firs_period': total_firs_period, 'resolved': resolved,
   448→        'resolution_rate': resolution_rate,
   449→        'category_stats': list(category_stats),
   450→        'monthly_trend': list(monthly_trend),
   451→        'priority_stats': list(priority_stats),
   452→        'total_fuel_cost': total_fuel_cost,
   453→        'vehicles_in_maintenance': vehicles_in_maintenance,
   454→        'page_title': 'Reports & Analytics',
   455→    }
   456→    return render(request, 'reports/reports.html', context)

   492→
   493→@login_required
   494→def vehicle_list_view(request):
   495→    vehicles = Vehicle.objects.all()
   496→    status_filter = request.GET.get('status', '')
   497→    type_filter = request.GET.get('type', '')
   498→    search_query = request.GET.get('q', '')
   499→
   500→    if status_filter:
   501→        vehicles = vehicles.filter(status=status_filter)
   502→    if type_filter:
   503→        vehicles = vehicles.filter(vehicle_type=type_filter)
   504→    if search_query:
   505→        vehicles = vehicles.filter(
   506→            Q(vehicle_number__icontains=search_query) |
   507→            Q(make__icontains=search_query) |
   508→            Q(model__icontains=search_query)
   509→        )
   510→
   511→    status_counts = {s: Vehicle.objects.filter(status=s).count() for s, _ in Vehicle.STATUS_CHOICES}
   512→
   513→    # Annotate each vehicle with its current assignee (if any)
   514→    from django.db.models import Subquery, OuterRef
   515→    active_assignments = VehicleAssignment.objects.filter(
   516→        vehicle=OuterRef('pk'), returned_date__isnull=True
   517→    ).select_related('officer').order_by('-assigned_date')
   518→
   519→    vehicles = vehicles.annotate(
   520→        _assignee_name=Subquery(

   607→def assign_vehicle(request, vehicle_id):
   608→    """Assign a vehicle to an officer for a specific shift/purpose."""
   609→    vehicle = get_object_or_404(Vehicle, pk=vehicle_id)
   610→    if vehicle.status == 'maintenance' or vehicle.status == 'out_of_service':
   611→        messages.error(request, f'Cannot assign {vehicle.vehicle_number} - it is currently {vehicle.get_status_display().lower()}.')
   612→        return redirect('vehicle_detail', vehicle_id=vehicle_id)
   613→
   614→    # Check if vehicle is already assigned and not returned
   615→    active_assignment = VehicleAssignment.objects.filter(vehicle=vehicle, returned_date__isnull=True).first()
   616→    if active_assignment:
   617→        messages.warning(request, f'{vehicle.vehicle_number} is already assigned to {active_assignment.officer.full_name}. Return it first before reassigning.')
   618→        return redirect('vehicle_detail', vehicle_id=vehicle_id)
   619→
   620→    form = VehicleAssignmentForm(request.POST)
   621→    if form.is_valid():
   622→        assignment = form.save(commit=False)
   623→        assignment.vehicle = vehicle
   624→        assignment.save()
   625→        # Update vehicle status
   626→        vehicle.status = 'in_use'
   627→        vehicle.mileage = assignment.odometer_start or vehicle.mileage
   628→        vehicle.save()
   629→        messages.success(request, f'{vehicle.vehicle_number} assigned to {assignment.officer.full_name} successfully.')
   630→    else:
   631→        for field, errors in form.errors.items():
   632→            for error in errors:
   633→                messages.error(request, f'{field}: {error}')
   634→    return redirect('vehicle_detail', vehicle_id=vehicle_id)
