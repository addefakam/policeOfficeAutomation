   382→   382→
   383→   383→@login_required
   384→   384→def reports_view(request):
   385→   385→    report_type = request.GET.get('type', 'overview')
   386→   386→    period = request.GET.get('period', 'month')
   387→   387→
   388→   388→    today = timezone.now().date()
   389→   389→    if period == 'week':
   390→   390→        start_date = today - timedelta(days=7)
   391→   391→    elif period == 'month':
   392→   392→        start_date = today - timedelta(days=30)
   393→   393→    elif period == 'quarter':
   394→   394→        start_date = today - timedelta(days=90)
   395→   395→    elif period == 'year':
   396→   396→        start_date = today - timedelta(days=365)
   397→   397→    else:
   398→   398→        start_date = today - timedelta(days=30)
   399→   399→
   400→   400→    firs_in_period = FIR.objects.filter(created_at__date__gte=start_date)
   401→   401→    total_firs_period = firs_in_period.count()
   402→   402→    resolved = firs_in_period.filter(status='closed').count()
   403→   403→    resolution_rate = round((resolved / total_firs_period * 100), 1) if total_firs_period > 0 else 0
   404→   404→
   405→   405→    category_stats = firs_in_period.values('crime_category').annotate(
   406→   406→        count=Count('id'),
   407→   407→        closed_count=Count('id', filter=Q(status='closed'))
   408→   408→    ).order_by('-count')
   409→   409→
   410→   410→    monthly_trend = FIR.objects.filter(
   411→   411→        created_at__date__gte=start_date
   412→   412→    ).annotate(
   413→   413→        month=TruncMonth('created_at')
   414→   414→    ).values('month').annotate(
   415→   415→        total=Count('id'),
   416→   416→        closed=Count('id', filter=Q(status='closed'))
   417→   417→    ).order_by('month')
   418→   418→
   419→   419→    priority_stats = firs_in_period.values('priority').annotate(count=Count('id')).order_by('-count')
   420→   420→
   421→   421→    avg_attendance = Attendance.objects.filter(date__gte=start_date, status='present').count()
   422→   422→    total_leave_days = LeaveRequest.objects.filter(
   423→   423→        status='approved', start_date__gte=start_date
   424→   424→    ).annotate(days=F('end_date') - F('start_date')).aggregate(total=Sum('days'))
   425→   425→
   426→   426→    total_fuel_cost = FuelLog.objects.filter(date__gte=start_date).aggregate(
   427→   427→        total=Sum(F('liters') * F('cost_per_liter'))
   428→   428→    )['total'] or 0
   429→   429→
   430→   430→    vehicles_in_maintenance = Vehicle.objects.filter(status='maintenance').count()
   431→   431→
   432→   432→    context = {
